To see combined directly in the database:
select web_markingcombinedresult.id, web_markingcombinedresult.marks, min(stu.gn), min(stu.fn), min(grad.id), string_agg(supervisor.dn,',')
from web_markingcombinedresult
join web_markingstudent stu
on stu.id=web_markingcombinedresult.student_id
join web_markinggrading grad
on grad.student_id=stu.id
join web_supervisor supervisor
on grad.marker_id=supervisor.id
where web_markingcombinedresult.marks is not null
group by web_markingcombinedresult.id
order by min(stu.gn),min(stu.fn);
